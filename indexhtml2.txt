/* Adatok betöltése */

const storageKey = "gyumolcsnaplo";

let data =
  JSON.parse(localStorage.getItem(storageKey) || "{}");


/* Adatok mentése */

function save() {

  localStorage.setItem(
    storageKey,
    JSON.stringify(data)
  );

}


/* Gyümölcs hozzáadása */

function addFruit(name) {

  if (!data[name]) {
    data[name] = 0;
  }

  data[name]++;

  save();

  render();

}


/* Mai nap törlése */

function resetDay() {

  if (
    confirm(
      "Biztosan törlöd a mai fogyasztást?"
    )
  ) {

    data = {};

    save();

    render();

  }

}


/* Emoji képpé alakítása */

function emojiImage(emoji) {

  const svg = `

  <svg
    xmlns="http://www.w3.org/2000/svg"
    width="240"
    height="240"
  >

    <rect
      width="100%"
      height="100%"
      rx="120"
      fill="#f4f7f2"
    />

    <text
      x="50%"
      y="55%"
      text-anchor="middle"
      dominant-baseline="middle"
      font-size="120"
    >
      ${emoji}
    </text>

  </svg>

  `;

  return (
    "data:image/svg+xml;charset=UTF-8," +
    encodeURIComponent(svg)
  );

}


/* Oldal frissítése */

function render() {

  const fruitsElement =
    document.getElementById("fruits");


  /* Gyümölcskártyák */

  fruitsElement.innerHTML =
    fruits.map(fruit => `

      <button
        class="fruit"
        onclick="addFruit('${fruit.name}')"
      >

        <img
          src="${emojiImage(fruit.emoji)}"
          alt="${fruit.name}"
        >

        <div class="fruit-name">
          ${fruit.name}
        </div>

        <div class="portion">
          kb. ${fruit.grams} g / adag
        </div>

        <div class="count">
          ${data[fruit.name] || 0} adag
        </div>

      </button>

    `).join("");


  /* Összes adag */

  const total =
    Object.values(data)
      .reduce(
        (sum, value) => sum + value,
        0
      );


  document.getElementById("total")
    .textContent = total;


  /* Haladási sáv */

  const percentage =
    Math.min(
      total / 5 * 100,
      100
    );

  document.getElementById("fill")
    .style.width =
      percentage + "%";


  /* Mai lista */

  const log =
    document.getElementById("log");


  const entries =
    Object.entries(data)
      .filter(
        ([name, count]) => count > 0
      );


  if (entries.length === 0) {

    log.innerHTML = `
      <div class="empty">
        Még nem választottál gyümölcsöt.
      </div>
    `;

  } else {

    log.innerHTML =
      entries.map(
        ([name, count]) => `

          <div class="today-row">

            <span>
              ${name}
            </span>

            <strong>
              ${count} adag
            </strong>

          </div>

        `
      ).join("");

  }

}


/* Első betöltés */

render();
